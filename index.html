return; } // بررسی ادمین if (u === 'admin' && p === 'admin123') { currentUser = { username: 'admin', role: 'admin' }; showAdminView(); return; } // استعلام مستقیم از پایگاه داده فایربیس db.ref('users/' + u).once('value') .then(snapshot => { const data = snapshot.val(); if (data && data.password === p) { currentUser = { username: u, ...data }; showUserView(); } else { alert('نام کاربری یا رمز عبور اشتباه است.'); } }) .catch(error => { alert('خطا در احراز هویت: ' + error.message); }); } function showUserView() { document.getElementById('view-login').classList.add('hidden'); document.getElementById('view-admin').classList.add('hidden'); document.getElementById('view-user').classList.remove('hidden'); document.getElementById('user-display-name').innerText = currentUser.fullName || currentUser.username; // اشتراک زنده بر روی تغییرات وضعیت کاربر در فایربیس db.ref('users/' + currentUser.username).on('value', snapshot => { const data = snapshot.val(); const checkbox = document.getElementById('user-approval-checkbox'); const btn = document.getElementById('btn-submit-approval'); if (data && data.approved) { checkbox.checked = true; checkbox.disabled = true; btn.disabled = true; btn.innerText = 'تاییدیه قبلاً ثبت شده است'; btn.style.backgroundColor = '#64748b'; } else { checkbox.checked = false; checkbox.disabled = false; btn.disabled = false; btn.innerText = 'ثبت نهایی تاییدیه'; btn.style.backgroundColor = 'var(--primary)'; } }); } function saveUserApproval() { const isChecked = document.getElementById('user-approval-checkbox').checked; if (!isChecked) { alert('لطفاً گزینه تایید را علامت بزنید.'); return; } const timestamp = new Date().toLocaleString('fa-IR'); db.ref('users/' + currentUser.username).update({ approved: true, approvalDate: timestamp }).then(() => { alert('تاییدیه شما با موفقیت در پایگاه داده ابری ثبت گردید.'); }).catch(error => { alert('خطا در ثبت: ' + error.message); }); } function showAdminView() { document.getElementById('view-login').classList.add('hidden'); document.getElementById('view-user').classList.add('hidden'); document.getElementById('view-admin').classList.remove('hidden'); loadAdminTable(); } function addUser() { const name = document.getElementById('new-name').value.trim(); const u = document.getElementById('new-username').value.trim(); const p = document.getElementById('new-password').value.trim(); if (!name || !u || !p) { alert('لطفاً تمامی فیلدهای نام، یوزرنیم و پسورد را تکمیل کنید.'); return; } db.ref('users/' + u).set({ fullName: name, password: p, approved: false, approvalDate: '-' }).then(() => { document.getElementById('new-name').value = ''; document.getElementById('new-username').value = ''; document.getElementById('new-password').value = ''; alert('کاربر جدید با موفقیت ایجاد شد و در تمام مرورگرها در دسترس است.'); }).catch(error => { alert('خطا در ذخیره‌سازی: ' + error.message); }); } function deleteUser(username) { if (confirm(`آیا از حذف کاربر ${username} اطمینان دارید؟`)) { db.ref('users/' + username).remove().catch(error => { alert('خطا در حذف کاربر: ' + error.message); }); } } function loadAdminTable() { // دریافت آنلاین و همگام فهرست کاربران db.ref('users').on('value', snapshot => { const users = snapshot.val() || {}; const tbody = document.getElementById('admin-table-body'); tbody.innerHTML = ''; const keys = Object.keys(users); if (keys.length === 0) { tbody.innerHTML = 'هیچ کاربری تعریف نشده است.'; return; } keys.forEach(key => { const val = users[key]; const tr = document.createElement('tr'); tr.innerHTML = ` ${val.fullName || '-'} ${key} ${val.approved ? 'تایید شده' : 'در انتظار تایید'} ${val.approvalDate || '-'} `; tbody.appendChild(tr); }); }); } function logout() { if (currentUser && currentUser.username !== 'admin') { db.ref('users/' + currentUser.username).off(); } db.ref('users').off(); currentUser = null; document.getElementById('login-username').value = ''; document.getElementById('login-password').value = ''; document.getElementById('view-login').classList.remove('hidden'); document.getElementById('view-user').classList.add('hidden'); document.getElementById('view-admin').classList.add('hidden'); }
